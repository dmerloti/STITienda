# login
@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        user = request.form['usuario']
        password = request.form['password']

        conn = get_db_connection()
        cursor = conn.cursor(dictionary=True)

        sql = """
            SELECT u.*, r.concepto AS rol
            FROM usuarios u
            JOIN usuarios_roles r ON u.idrol = r.id
            WHERE u.username = %s
        """
        cursor.execute(sql, (user,))
        user_db = cursor.fetchone()
        conn.close()

        if user_db and check_password_hash(user_db['password'], password):
            session['user'] = user_db['username']
            session['idrol'] = user_db['idrol']
            session['rol'] = user_db['rol']  # admin o vendedor
            return redirect('/')
        else:
            flash("Usuario o contraseña incorrectos - Reintente")
            return redirect('/login')

    return render_template('login.html')


# logout
@app.route('/logout')
def logout():
    session.clear()
    return redirect(url_for('tienda.index'))

# proteger rutas
def login_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if 'user' not in session:
            return redirect(url_for('login'))
        return f(*args, **kwargs)
    return decorated_function

# Decorador para ADMIN
def admin_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if 'user' not in session:
            return redirect('/login')
        if session.get('rol') != 'admin':
            return redirect('/')
        return f(*args, **kwargs)
    return decorated_function

@app.route('/usuarios/crear_usuario', methods=['GET', 'POST'])
@admin_required
def crear_usuario():

    conexion = get_db_connection()
    cursor = conexion.cursor(dictionary=True)

    if request.method == 'POST':
        username = request.form['username']
        password = generate_password_hash(request.form['password'])
        idrol = request.form['idrol']

        sql = "INSERT INTO usuarios (username, password, idrol) VALUES (%s, %s, %s)"
        cursor.execute(sql, (username, password, idrol))
        conexion.commit()

        conexion.close()
        return redirect('/usuarios')

    # Cargar roles para el combo
    cursor.execute("SELECT * FROM usuarios_roles")
    roles = cursor.fetchall()
    conexion.close()

    return render_template('usuarios/crear_usuario.html', roles=roles)

@app.route('/usuarios/editar_usuario/<int:id>', methods=['GET', 'POST'])
@admin_required
def editar_usuario(id):

    conexion = get_db_connection()
    cursor = conexion.cursor(dictionary=True)

    if request.method == 'POST':
        username = request.form['username']
        password = request.form['password']
        idrol = request.form['idrol']

        if password:
            password_hash = generate_password_hash(password)
            sql = "UPDATE usuarios SET username=%s, password=%s, idrol=%s WHERE id=%s"
            cursor.execute(sql, (username, password_hash, idrol, id))
        else:
            sql = "UPDATE usuarios SET username=%s, idrol=%s WHERE id=%s"
            cursor.execute(sql, (username, idrol, id))

        conexion.commit()
        conexion.close()

        return redirect('/usuarios')

    # Traer usuario
    cursor.execute("SELECT * FROM usuarios WHERE id=%s", (id,))
    usuario = cursor.fetchone()

    # Traer roles
    cursor.execute("SELECT * FROM usuarios_roles")
    roles = cursor.fetchall()

    conexion.close()

    return render_template('usuarios/editar_usuario.html',
                           usuario=usuario,
                           roles=roles)

@app.route('/registro', methods=['GET', 'POST'])
def registro():
    if request.method == 'POST':
        username = request.form['username']
        password = generate_password_hash(request.form['password'])

        conexion = get_db_connection()
        cursor = conexion.cursor(dictionary=True)

        # Buscar ID del rol cliente
        cursor.execute("SELECT id FROM usuarios_roles WHERE concepto = %s", ('cliente',))
        rol_cliente = cursor.fetchone()

        if not rol_cliente:
            flash("No existe el rol cliente en la base de datos.")
            conexion.close()
            return redirect('/registro')

        idrol = rol_cliente['id']

        # Insertar usuario nuevo
        sql = "INSERT INTO usuarios (username, password, idrol) VALUES (%s, %s, %s)"
        cursor.execute(sql, (username, password, idrol))
        conexion.commit()
        conexion.close()

        flash("Usuario creado correctamente.")
        return redirect(url_for('login'))

    return render_template('usuarios/registro.html')

@app.route('/guardar_articulo', methods=['POST'])
@login_required
def guardar_articulo():
    _codigo = request.form['txtCodigo']
    _concepto = request.form['txtConcepto']
    _foto = request.files['txtFoto']

    if _codigo == '' or _concepto == '' or _foto.filename == '':
        flash('Recuerda llenar todos los campos')
        return redirect(url_for('create_articulo'))

    now = datetime.now()
    tiempo = now.strftime("%Y%m%d%H%M%S")

    filename = secure_filename(_foto.filename)
    nuevoNombreFoto = f"{tiempo}_{filename}"

    ruta_foto = os.path.join(app.config['CARPETA'], nuevoNombreFoto)
    _foto.save(ruta_foto)

    conn = get_db_connection()
    cursor = conn.cursor()
    sql = """
        INSERT INTO articulos (codigo, concepto, foto)
        VALUES (%s, %s, %s)
    """
    cursor.execute(sql, (_codigo, _concepto, nuevoNombreFoto))
    conn.commit()

    cursor.close()
    conn.close()

    return redirect(url_for('articulos'))

@app.route('/actualizar_articulo', methods=["POST"])
@login_required
def update():
    id = request.form['txtID']
    _codigo = request.form['txtCodigo']
    _concepto = request.form['txtConcepto']
    _foto = request.files['txtFoto']

    conn = get_db_connection()
    cursor = conn.cursor(dictionary=True)

    # Si hay foto nueva
    if _foto and _foto.filename != '':
        now = datetime.now()
        tiempo = now.strftime("%Y%H%M%S")
        nuevoNombreFoto = tiempo + _foto.filename
        _foto.save(os.path.join(app.config['CARPETA'], nuevoNombreFoto))

        # Borrar foto anterior
        cursor.execute("SELECT foto FROM articulos WHERE id=%s", (id,))
        fila = cursor.fetchone()
        if fila and fila['foto']:
            ruta_foto = os.path.join(app.config['CARPETA'], fila['foto'])
            if os.path.exists(ruta_foto):
                os.remove(ruta_foto)

        sql = """
            UPDATE articulos
            SET codigo=%s, concepto=%s, foto=%s
            WHERE id=%s
        """
        datos = (_codigo, _concepto, nuevoNombreFoto, id)

    # Si NO se cambia la foto
    else:
        sql = """
            UPDATE articulos
            SET codigo=%s, concepto=%s
            WHERE id=%s
        """
        datos = (_codigo, _concepto, id)

    cursor.execute(sql, datos)
    conn.commit()
    cursor.close()
    conn.close()

    return redirect(url_for('articulos'))
